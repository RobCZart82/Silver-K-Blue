cmake_minimum_required(VERSION 3.22)
juce_add_binary_data(silverData SOURCES ../../kstation/kstationJucePlugin/skins/silver-k.json ../../kstation/kstationJucePlugin/skins/silver-k-knob-body.png ../../kstation/kstationJucePlugin/skins/silver-k-knob-grip.png ../../kstation/kstationJucePlugin/skins/silver-k-stand-knob-base.png ../../kstation/kstationJucePlugin/skins/silver-k-button-off.png ../../kstation/kstationJucePlugin/skins/silver-k-button-on.png ../../../../assets/ks-init-flash.bin)
set(SILVER_FORMATS VST3 Standalone)
if(APPLE)
 list(APPEND SILVER_FORMATS AU)
endif()
juce_add_plugin(silverK VERSION 0.2.4 COMPANY_NAME "The Usual Suspects" BUNDLE_ID "com.theusualsuspects.silverk" PLUGIN_MANUFACTURER_CODE TusP PLUGIN_CODE Tskv PRODUCT_NAME "Silver-K" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_AUTO_MANIFEST FALSE FORMATS ${SILVER_FORMATS})
target_sources(silverK PRIVATE silverProcessor.cpp silverProcessor.h silverDataFolder.cpp silverPresets.cpp silverEditor.cpp silverEditor.h silverLibrary.cpp silverLibrary.h silverPatch.cpp silverPatch.h silverVStation.cpp silverVStation.h silverKEngine.cpp silverKEngine.h silverBrowser.cpp silverBrowser.h silverClock.h silverMidiLearn.h silverPerformance.cpp silverPerformance.h)
target_compile_definitions(silverK PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(silverK PRIVATE kstationLib ksLib silverData jucePluginLib juce::juce_audio_utils juce::juce_audio_plugin_client)
# Standalone's own JUCE module objects must use the same optional ASIO setting
# as the shared juce_plugin_modules target.
if(WIN32 AND SilverK_ASIO_SDK_PATH AND EXISTS "${SilverK_ASIO_SDK_PATH}/common/iasiodrv.h")
 target_compile_definitions(silverK PUBLIC JUCE_ASIO=1)
 target_include_directories(silverK PUBLIC "${SilverK_ASIO_SDK_PATH}/common")
endif()
if(APPLE)
foreach(format IN LISTS SILVER_FORMATS)
 target_sources(silverK_${format} PRIVATE "${ROOT}/LICENSE.md")
 set_source_files_properties("${ROOT}/LICENSE.md" PROPERTIES HEADER_FILE_ONLY TRUE MACOSX_PACKAGE_LOCATION Resources)
 add_custom_command(TARGET silverK_${format} POST_BUILD COMMAND /usr/bin/codesign --force --deep --sign - --timestamp=none "$<TARGET_BUNDLE_DIR:silverK_${format}>" VERBATIM)
endforeach()
endif()
enable_testing()
add_executable(silver_check silver_check.cpp)
target_link_libraries(silver_check PRIVATE silverK jucePluginLib ksLib kstationLib juce::juce_audio_utils)
if(WIN32)
 target_link_libraries(silver_check PRIVATE dbghelp)
endif()
add_test(NAME silver_conversion COMMAND silver_check)

add_executable(silver_vst3_check vst3_check.cpp)
target_compile_definitions(silver_vst3_check PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(silver_vst3_check PRIVATE juce::juce_audio_utils)
add_dependencies(silver_vst3_check silverK_VST3)
if(APPLE)
add_executable(silver_au_check au_check.cpp)
target_compile_definitions(silver_au_check PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(silver_au_check PRIVATE juce::juce_audio_utils "-framework AudioToolbox")
add_dependencies(silver_au_check silverK_AU)
endif()

add_executable(silverClock_test silverClock_test.cpp)
target_compile_features(silverClock_test PRIVATE cxx_std_17)
add_test(NAME silver_host_clock COMMAND silverClock_test)

target_link_libraries(silverWorkflow_test PRIVATE silverK jucePluginLib ksLib kstationLib juce::juce_audio_utils)
add_test(NAME silver_patch_workflow COMMAND silverWorkflow_test)
add_executable(silverEditor_test silverEditor_test.cpp)
target_link_libraries(silverEditor_test PRIVATE silverK jucePluginLib ksLib kstationLib juce::juce_audio_utils)
add_test(NAME silver_native_editor COMMAND silverEditor_test)
add_executable(silverRealtime_test EXCLUDE_FROM_ALL silverRealtime_test.cpp)
target_link_libraries(silverRealtime_test PRIVATE silverK jucePluginLib ksLib kstationLib juce::juce_audio_utils)
add_executable(silverPerformance_test silverPerformance_test.cpp)
target_link_libraries(silverPerformance_test PRIVATE silverK jucePluginLib ksLib kstationLib juce::juce_audio_utils)
add_test(NAME silver_performance_editor COMMAND silverPerformance_test)

add_executable(silverPersistence_test silverPersistence_test.cpp)
target_link_libraries(silverPersistence_test PRIVATE silverK jucePluginLib ksLib kstationLib juce::juce_audio_utils)
add_test(NAME silver_persistence_and_rescan COMMAND silverPersistence_test)
